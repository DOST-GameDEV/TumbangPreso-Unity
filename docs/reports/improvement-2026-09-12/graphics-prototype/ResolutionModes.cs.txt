using UnityEngine;

namespace TumbangPreso.Settings
{
    /// <summary>World pixel density, independent of the native-resolution interface.</summary>
    public static class ResolutionModes
    {
        public static readonly float[] Scales = { .5f, 2f/3f, 5f/6f, 1f, 1.25f, 1.5f };
        public static readonly string[] Labels =
            { "50%", "67%", "83%", "100% (Native)", "125% (Supersampling)", "150% (Supersampling)" };
        public const int Default = 3;
        public static int Current { get; private set; } = Default;
        public static float Scale => Scales[Current];
        public static void Apply(int index) => Current = Mathf.Clamp(index, 0, Scales.Length - 1);
    }
}
